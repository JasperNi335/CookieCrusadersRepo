cmake_minimum_required(VERSION 3.21)

# Define the project
project(Pico_w)

# Initialize the Pico SDK
include(pico_sdk_import.cmake)

# Set the target platform for the Raspberry Pi Pico
set(PICO_BOARD pico_w)

# Initialize the SDK and set up the project
pico_sdk_init()

# Add the executable that will be built
add_executable(Pico_w main.c sound_sensor.c)

# Include the directory where header files are located
target_include_directories(Pico_w PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include)

# Link the Raspberry Pi Pico SDK and libraries
target_link_libraries(Pico_w pico_stdlib hardware_gpio)

# Create a USB serial interface for debugging
pico_add_extra_outputs(Pico_w)
